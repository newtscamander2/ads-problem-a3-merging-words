\usepackage{lptest}

  { \str_if_eq:eeTF { \use:c { latexpro@wavefactor } } { 2 } { \iftrue } { \iffalse } }

\lptrue{box~option~wins~over~the~package~option}
  { \str_if_eq:eeTF { \use:c { latexpro@edgestyle } } { continued } { \iftrue } { \iffalse } }
